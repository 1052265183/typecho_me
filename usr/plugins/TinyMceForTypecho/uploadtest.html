// JavaScript上传函数修改 async function uploadFile(file) { const uploadId = 'upload_' + Date.now() + '_' + Math.random().toString(36).substr(2); const formData = new FormData(); formData.append('file', file); formData.append('upload_id', uploadId); // 添加upload_id到表单 // 使用XMLHttpRequest以便设置请求头 const xhr = new XMLHttpRequest(); // 监听上传进度（客户端端进度） xhr.upload.addEventListener('progress', function(e) { if (e.lengthComputable) { const percentComplete = (e.loaded / e.total) * 100; console.log('客户端进度:', percentComplete + '%'); } }); return new Promise((resolve, reject) => { xhr.onload = function() { if (xhr.status === 200) { try { const response = JSON.parse(xhr.responseText); if (response.success) { resolve(response); } else { reject(new Error(response.error || '上传失败')); } } catch (e) { reject(new Error('解析响应失败: ' + e.message)); } } else { reject(new Error('HTTP错误: ' + xhr.status)); } }; xhr.onerror = function() { reject(new Error('网络错误')); }; // 启动服务器端进度轮询 if (UploadConfig.ENABLE_PROGRESS) { startServerProgressPolling(uploadId, file.size); } xhr.open('POST', CONFIG.uploadUrl); xhr.setRequestHeader('X-Requested-With', 'XMLHttpRequest'); xhr.setRequestHeader('X-Upload-ID', uploadId); xhr.send(formData); }); } // 启动服务器进度轮询 function startServerProgressPolling(uploadId, totalSize) { let progressInterval = null; const fetchProgress = async () => { try { const response = await fetch(`${CONFIG.progressUrl}?action=progress&upload_id=${uploadId}`); const data = await response.json(); if (data.success && data.data.progress) { const progress = data.data.progress; // 更新UI updateProgressUI({ percentage: progress.percentage, uploaded: progress.uploaded, total: totalSize, speed: progress.speed, status: progress.status }); // 如果完成或失败，停止轮询 if (progress.status === 'complete' || progress.status === 'failed') { if (progressInterval) { clearInterval(progressInterval); } } } } catch (error) { console.error('获取进度失败:', error); } }; // 立即获取一次 fetchProgress(); // 设置定时器 progressInterval = setInterval(fetchProgress, 500); return progressInterval; }
